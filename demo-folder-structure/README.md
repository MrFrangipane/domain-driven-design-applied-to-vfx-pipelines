    │                   │   └── an-awesome-show_character_joe_design_v001.psd
    │                   └── v002/
    │                       └── an-awesome-show_character_joe_design_v002.psd
    └── sequences/
        └── sq001/
            └── shots/
                └── sh001/
                    ├── animation/
                    │   ├── publish/
                    │   │   ├── v001/
                    │   │   │   └── an-awesome-show_sq001_sh001_animation_v001.ma
                    │   │   ├── v002/

                    └── layout/
                        ├── publish/
                        │   ├── v001/

                        │   ├── v002/
                        │   │   └── an-awesome-show_sq001_sh001_layout_v002.ma

                        │       └── an-awesome-show_sq001_sh001_layout_v003.ma
